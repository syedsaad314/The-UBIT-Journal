import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {

        String file = "C://Users//wtc//Desktop//EVE01Sales.txt";
        HashMap<String, Integer> productWise = new HashMap<String, Integer>();

        try {
            FileReader fr = new FileReader(file);
            BufferedReader br = new BufferedReader(fr);
            String line = br.readLine();  //skip headings

            while ((line = br.readLine()) != null) {
                String[] parts = line.split("\t");

                String product = parts[3];
                int qty = Integer.parseInt(parts[4]);
                int unitPrice = Integer.parseInt(parts[5]);
                int sale = qty * unitPrice;

                if (productWise.containsKey(product))
                    productWise.put(product, productWise.get(product) + sale);
                else
                    productWise.put(product, sale);

            }

            System.out.println("Product-wise Sum:");
            for(String product : productWise.keySet()) {
                System.out.println(product + ": $" + productWise.get(product));
            }
            br.close();

        } 
        catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }
}
