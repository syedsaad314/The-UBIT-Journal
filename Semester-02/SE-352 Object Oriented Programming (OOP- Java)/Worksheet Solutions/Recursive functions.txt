Fibonacci Algorithm
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        int n;
        System.out.print("Enter a number:");
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        if(n<1)
            System.out.println("Invalid input");
        for(int i=1;i<=n;i++) {
            System.out.print(fib(i) + " ");
        }
    }

    public static int fib(int n) {
        if(n==1 || n==2)
            return 1;
        return fib(n-1)+fib(n-2);
    }
}

GCD
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        int n, m;
        System.out.print("Enter two numbers:");
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        System.out.println("GCD: " + gcd(n,m));
    }

    public static int gcd(int n, int m) {
        if(n==0)
            return m;
        if(m==0)
            return n;
        return gcd(m, n%m);
    }
}

Max of an array
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        int n;
        System.out.print("Enter size of array:");
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int[] arr = new int[n];
        System.out.print("Enter array elements:");
        for(int i=0; i<n; i++) {
            arr[i] = sc.nextInt();
        }
        int max = arr[0];
        System.out.println("Max val: " + max(arr,n-1,max));
    }

    public static int max(int[] arr, int n, int max) {
        if(n<0)
            return max;
        if(arr[n]>max)
            max=arr[n];
        return max(arr, n-1, max);
    }
}

Consecutive Addition
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        int start, end;
        System.out.println("Enter starting number:");
        Scanner sc = new Scanner(System.in);
        start = sc.nextInt();
        System.out.println("Enter ending number:");
        end = sc.nextInt();
        System.out.println("Sum: " + add(start, end));
    }

    public static int add(int start, int end) {
        if(start>end)
            return 0;
        return start + add(start+1, end);
    }
}
