public class bookInfo {

    public String bookTitle, author; public int noOfPages, cost;
    private static int noOfBooks = 0; //class level integer, we made it private so that the user can't change it

    public bookInfo() {  //everytime bookInfo is accessed, a new book is added to the system
        noOfBooks += 1; }

    int getNoOfBooks() {
        return noOfBooks; }

    public bookInfo(String bookTitle, String author, int noOfPages, int cost) {  //method overloading
        noOfBooks += 1;
        this.bookTitle = bookTitle;
        this.author = author;
        this.noOfPages = noOfPages;
        this.cost = cost;
    }

    //incase the book was donated, it has no cost, so:
    public bookInfo(String bookTitle, String author, int noOfPages) {
        this (bookTitle, author, noOfPages, 0);
    }
}

class ebookInfo extends bookInfo { //inheritance
    public String folderPath;

    public ebookInfo(String ebookTitle, String ebookAuthor, int ebookNoOfPages, int ebookCost, String folderPath) {
        super(ebookTitle, ebookAuthor, ebookNoOfPages, ebookCost);
        this.folderPath = folderPath;
    }
}
